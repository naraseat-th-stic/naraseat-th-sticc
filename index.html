ยินดีต้อนรับ นายนราเศรษฐ์ ธัญศิรอนันต์

ระบบนี้ออกแบบมาเพื่อช่วยจดบันทึกและติดตามงานทุกขั้นตอน ใช้งานง่ายใน 3 ขั้นตอน:

1. สร้างใบงาน: กดที่เมนู "สร้างใบงานใหม่" ระบุหัวข้องาน วันเริ่มงาน วันครบกำหนด และผู้มอบหมายงาน จากนั้นกดบันทึก
2. ตรวจสอบงานประจำวันและสัปดาห์: ดูที่เมนู "ติดตามงานรายสัปดาห์" ท่านจะเห็นทันทีว่าวันนี้ต้องทำอะไรบ้าง และแต่ละวันในสัปดาห์มีกำหนดส่งงานใด
3. ระบบแจ้งเตือนงานล่าช้าอัตโนมัติ:
  • หากงานเกินกำหนด 3 วัน: จะขึ้นแถบเตือนสีส้ม ⚠️
  • หากงานเกินกำหนด 5 วันขึ้นไป: จะขึ้นแถบเตือนสีแดงกระพริบ 🚨 เพื่อให้ท่านเร่งรัดได้ทันเวลา

* ทุกข้อมูลจะถูกบันทึกไว้ในคอมพิวเตอร์เครื่องนี้โดยอัตโนมัติ และท่านสามารถกดส่งออกเป็น Excel ได้ทุกเวลา

📊 วิธีเชื่อมต่อ Google Sheets เป็น Database ใน 3 นาที

ทำตาม 4 ขั้นตอนง่ายๆ ด้านล่างนี้เพียงครั้งเดียว ท่านก็จะได้ Google Sheets ที่บันทึกข้อมูลใบงานแบบเรียลไทม์ ปลอดภัย และอยู่ตลอดไป:

ขั้นตอนที่ 1: เปิด Google Sheets ใหม่

เข้าเว็บ sheets.google.com แล้วกดสร้างสเปรดชีตเปล่าใหม่ 1 ไฟล์ (ตั้งชื่อไฟล์ว่า "ระบบติดตามงาน นายนราเศรษฐ์")

ขั้นตอนที่ 2: วางโค้ด Apps Script

ที่เมนูด้านบนของ Google Sheets คลิก ส่วนขยาย (Extensions) ➔ Apps Script

ลบข้อความเดิมในหน้านั้นออกทั้งหมด แล้วคัดลอกโค้ดด้านล่างนี้ไปวางแทนที่:

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  var tickets = [];
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    if (!row[0]) continue;
    tickets.push({
      id: String(row[0]),
      title: String(row[1] || ''),
      description: String(row[2] || ''),
      assigner: String(row[3] || ''),
      category: String(row[4] || ''),
      priority: String(row[5] || 'normal'),
      startDate: String(row[6] || ''),
      endDate: String(row[7] || ''),
      duration: String(row[8] || ''),
      status: String(row[9] || 'pending'),
      progress: Number(row[10] || 0),
      notes: String(row[11] || ''),
      completedAt: String(row[12] || ''),
      createdAt: String(row[13] || '')
    });
  }
  return ContentService.createTextOutput(JSON.stringify(tickets)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var rawData = e.postData.contents;
    var parsed = JSON.parse(rawData);
    var tickets = Array.isArray(parsed) ? parsed : (parsed.tickets || []);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    sheet.clearContents();
    var headers = ['รหัสใบงาน', 'หัวข้องาน', 'รายละเอียด', 'ผู้มอบหมาย', 'หมวดหมู่', 'ความสำคัญ', 'วันเริ่มงาน', 'วันสิ้นสุดงาน', 'ระยะเวลา', 'สถานะ', 'ความคืบหน้า%', 'บันทึกช่วยจำ', 'วันที่เสร็จจริง', 'วันที่สร้าง'];
    var outputRows = [headers];
    for (var i = 0; i < tickets.length; i++) {
      var t = tickets[i];
      outputRows.push([
        t.id, t.title, t.description || '', t.assigner, t.category || '', t.priority,
        t.startDate, t.endDate, t.duration || '', t.status, t.progress || 0,
        t.notes || '', t.completedAt || '', t.createdAt || ''
      ]);
    }
    sheet.getRange(1, 1, outputRows.length, headers.length).setValues(outputRows);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#2563eb').setFontColor('#ffffff');
    return ContentService.createTextOutput(JSON.stringify({ success: true, count: tickets.length })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

ขั้นตอนที่ 3: เผยแพร่เว็บแอป (Deploy)

  1. กดปุ่มสีน้ำเงิน "ทำให้ใช้งานได้" (Deploy) ที่มุมขวาบน ➔ เลือก "การทำให้ใช้งานได้ใหม่" (New deployment)
  2. กดไอคอนรูปฟันเฟือง ⚙️ ทางซ้าย เลือกประเภทเป็น "เว็บแอป" (Web app)
  3. ช่อง ผู้มีสิทธิ์เข้าถึง (Who has access): ให้เปลี่ยนเป็น "ทุกคน" (Anyone) (สำคัญมาก เพื่อให้ส่งข้อมูลได้)
  4. กดปุ่ม "ทำให้ใช้งานได้" (Deploy) แล้วกดอนุญาตสิทธิ์ (Authorize access) ให้เรียบร้อย

ขั้นตอนที่ 4: นำลิงก์มาวางในโปรแกรม

คัดลอก URL เว็บแอป (Web app URL) ที่ได้มา วางลงในช่อง "ลิงก์ Google Apps Script" ในแท็บตั้งค่า แล้วกด "บันทึกและทดสอบเชื่อมต่อ" ข้อมูลจะถูกซิงค์ขึ้น Google Sheets ทันที!
